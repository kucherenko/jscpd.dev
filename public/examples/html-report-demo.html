  def total(lines, opts \\ []) do
    currency = Keyword.get(opts, :currency, "EUR")

    lines
    |> Enum.reject(&(&1.quantity == 0))
    |> Enum.map(fn line -> line.unit_price * line.quantity end)
    |> Enum.sum()
    |> round_to_cents()
    |> then(&{currency, &1})
  end
